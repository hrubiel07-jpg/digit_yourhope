Chemin du fichier : htdocs/sync_data.php

Ce fichier recevra les données de l'application Android et les insérera dans la base de données.

PHP
<?php
// Connexion à la base de données (À remplir avec vos accès InfinityFree)
$host = "sqlxxx.infinityfree.com";
$user = "if0_xxxxxx";
$pass = "votre_mot_de_passe";
$db   = "if0_xxxxxx_gs_services_db";

$conn = new mysqli($host, $user, $pass, $db);

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    // Récupération des données JSON envoyées par l'application Android
    $json = file_get_contents('php://input');
    $data = json_decode($json, true);

    if ($data) {
        $stmt = $conn->prepare("INSERT INTO rapports_taxis (proprietaire_id, recette_brute, depenses_pannes) VALUES (?, ?, ?)");
        $stmt->bind_param("idd", $data['id'], $data['recette'], $data['depenses']);
        
        if ($stmt->execute()) {
            echo json_encode(["status" => "success", "message" => "Données synchronisées"]);
        } else {
            echo json_encode(["status" => "error", "message" => $conn->error]);
        }
    }
}
?>

$dbname = "ton_nom_de_bdd";

// Réception des données de l'application Android
$json = file_get_contents('php://input');
$data = json_decode($json, true);

if ($data) {
    $proprio = $data['nom_proprietaire'];
    $recette = $data['recette_mensuelle'];

<?php
include 'db_connect.php';

header('Content-Type: application/json');

// Vérifier la méthode HTTP
if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    echo json_encode(['status' => 'error', 'message' => 'Méthode non autorisée']);
    exit;
}

// Récupérer les données JSON
$json = file_get_contents('php://input');
$data = json_decode($json, true);

if (!$data) {
    echo json_encode(['status' => 'error', 'message' => 'Données JSON invalides']);
    exit;
}

// Valider les données requises
$required_fields = ['code_proprietaire', 'nom_proprietaire', 'recette_mensuelle'];
foreach ($required_fields as $field) {
    if (!isset($data[$field])) {
        echo json_encode(['status' => 'error', 'message' => 'Champ manquant: ' . $field]);
        exit;
    }
}

// Préparer les valeurs avec défauts
$code = $conn->real_escape_string($data['code_proprietaire']);
$nom = $conn->real_escape_string($data['nom_proprietaire']);
$immat = isset($data['immatriculation']) ? $conn->real_escape_string($data['immatriculation']) : 'Non spécifiée';
$invest = isset($data['investissement_total']) ? floatval($data['investissement_total']) : 0;
$recette = floatval($data['recette_mensuelle']);
$depenses = isset($data['depenses_pannes']) ? floatval($data['depenses_pannes']) : 0;
$commission = isset($data['commission_gs']) ? floatval($data['commission_gs']) : ($recette * 0.10); // 10% par défaut

// Vérifier si le rapport existe déjà pour ce mois
$sql_check = "SELECT id FROM rapports WHERE code_proprietaire = '$code' 
              AND MONTH(date_rapport) = MONTH(CURRENT_DATE())
              AND YEAR(date_rapport) = YEAR(CURRENT_DATE())";
$result = $conn->query($sql_check);

if ($result->num_rows > 0) {
    // Mettre à jour le rapport existant
    $sql = "UPDATE rapports SET 
            recette_mensuelle = '$recette',
            depenses_pannes = '$depenses',
            commission_gs = '$commission'
            WHERE code_proprietaire = '$code' 
            AND MONTH(date_rapport) = MONTH(CURRENT_DATE())";
} else {
    // Insérer un nouveau rapport
    $sql = "INSERT INTO rapports 
            (code_proprietaire, nom_proprietaire, immatriculation, 
             investissement_total, recette_mensuelle, depenses_pannes, commission_gs) 
            VALUES 
            ('$code', '$nom', '$immat', '$invest', '$recette', '$depenses', '$commission')";
}

if ($conn->query($sql) === TRUE) {
    echo json_encode([
        'status' => 'success', 
        'message' => 'Données synchronisées avec succès',
        'id' => $conn->insert_id
    ]);
} else {
    echo json_encode([
        'status' => 'error', 
        'message' => 'Erreur lors de la synchronisation: ' . $conn->error
    ]);
}

$conn->close();
?>

package com.digitalyourhope.gsservices

import android.os.Bundle
import android.widget.Button
import android.widget.ProgressBar
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONObject
import java.io.IOException

class SyncActivity : AppCompatActivity() {

    private lateinit var progressBar: ProgressBar
    private lateinit var txtStatus: TextView
    private lateinit var btnSync: Button

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_sync)

        progressBar = findViewById(R.id.progressBar)
        txtStatus = findViewById(R.id.txtStatus)
        btnSync = findViewById(R.id.btnSync)

        btnSync.setOnClickListener {
            synchroniserDonnees()
        }
    }

    private fun synchroniserDonnees() {
        // Simulation de données (à remplacer par vos vraies données)
        val rapport = JSONObject().apply {
            put("code_proprietaire", "GS001")
            put("nom_proprietaire", "Jean Dupont")
            put("immatriculation", "TV-889-AB")
            put("investissement_total", 4500000)
            put("recette_mensuelle", 300000)
            put("depenses_pannes", 25000)
            put("commission_gs", 30000)
        }

        CoroutineScope(Dispatchers.IO).launch {
            envoyerDonneesAuServeur(rapport.toString())
        }
    }

    private suspend fun envoyerDonneesAuServeur(jsonData: String) {
        withContext(Dispatchers.Main) {
            progressBar.visibility = android.view.View.VISIBLE
            txtStatus.text = "Synchronisation en cours..."
        }

        val client = OkHttpClient()
        val mediaType = "application/json; charset=utf-8".toMediaType()
        val body = jsonData.toRequestBody(mediaType)

        val request = Request.Builder()
            .url("https://gsservices.page.gd/sync.php")
            .post(body)
            .build()

        try {
            val response = client.newCall(request).execute()
            val responseBody = response.body?.string()

            withContext(Dispatchers.Main) {
                progressBar.visibility = android.view.View.GONE
                
                if (response.isSuccessful && responseBody != null) {
                    val jsonResponse = JSONObject(responseBody)
                    if (jsonResponse.getString("status") == "success") {
                        txtStatus.text = "✅ Synchronisation réussie!"
                        txtStatus.setTextColor(resources.getColor(android.R.color.holo_green_dark))
                    } else {
                        txtStatus.text = "❌ Erreur: ${jsonResponse.getString("message")}"
                        txtStatus.setTextColor(resources.getColor(android.R.color.holo_red_dark))
                    }
                } else {
                    txtStatus.text = "❌ Erreur de connexion au serveur"
                    txtStatus.setTextColor(resources.getColor(android.R.color.holo_red_dark))
                }
            }
        } catch (e: IOException) {
            withContext(Dispatchers.Main) {
                progressBar.visibility = android.view.View.GONE
                txtStatus.text = "❌ Pas de connexion Internet"
                txtStatus.setTextColor(resources.getColor(android.R.color.holo_red_dark))
            }
        }
    }
}

// CONFIGURATION ADAPTATIVE - DIGITAL YOURHOPE

// Détection si on est en local (XAMPP) ou en ligne (InfinityFree)
function isLocalhost() {
    return $_SERVER['HTTP_HOST'] == 'localhost' 
        || $_SERVER['HTTP_HOST'] == '127.0.0.1'
        || strpos($_SERVER['HTTP_HOST'], 'localhost') !== false;
}

if (isLocalhost()) {
    // CONFIGURATION POUR XAMPP (EN LOCAL)
    define('DB_HOST', 'localhost');
    define('DB_USER', 'root');
    define('DB_PASS', '');  // Mot de passe vide par défaut sur XAMPP
    define('DB_NAME', 'gs_services_db');
} else {
    // CONFIGURATION POUR INFINITYFREE (EN LIGNE)
    define('DB_HOST', 'sqlxxx.infinityfree.com');  // Remplace par ton hostname réel
    define('DB_USER', 'if0_xxxxxx');              // Remplace par ton utilisateur
    define('DB_PASS', 'ton_mot_de_passe');        // Remplace par ton mot de passe
    define('DB_NAME', 'if0_xxxxxx_gs_services_db'); // Remplace par ta base

package com.digitalyourhope.gsservices.utils

import android.content.ContentValues
import android.content.Context
import android.database.sqlite.SQLiteDatabase
import android.database.sqlite.SQLiteOpenHelper
import com.digitalyourhope.gsservices.model.Rapport
import com.digitalyourhope.gsservices.model.Chauffeur

class DatabaseHelper(context: Context) : SQLiteOpenHelper(context, DATABASE_NAME, null, DATABASE_VERSION) {

    companion object {
        private const val DATABASE_NAME = "gsservices.db"
        private const val DATABASE_VERSION = 1
        
        // Table Rapports
        const val TABLE_RAPPORTS = "rapports"
        const val COL_ID = "id"
        const val COL_CODE = "code_proprietaire"
        const val COL_NOM = "nom_proprietaire"
        const val COL_IMMAT = "immatriculation"
        const val COL_INVESTISSEMENT = "investissement_total"
        const val COL_RECETTE = "recette_mensuelle"
        const val COL_DEPENSES = "depenses_pannes"
        const val COL_SALAIRE = "salaire_chauffeur"
        const val COL_COMMISSION = "commission_gs"
        const val COL_RECETTE_NETTE = "recette_nette"
        const val COL_DATE = "date_rapport"
        const val COL_SYNC = "est_synchronise"
        
        // Table Chauffeurs
        const val TABLE_CHAUFFEURS = "chauffeurs"
        const val COL_CHAUFFEUR_ID = "id"
        const val COL_NOM_CHAUFFEUR = "nom"
        const val COL_TELEPHONE = "telephone"
        const val COL_PERMIS = "numero_permis"
        const val COL_PHOTO_PERMIS = "photo_permis"
        const val COL_VEHICULE = "vehicule_id"
        const val COL_DATE_EMBAUCHE = "date_embauche"
        const val COL_SALAIRE_BASE = "salaire_base"
        const val COL_PRET_EN_COURS = "pret_en_cours"
        const val COL_ACTIF = "est_actif"
    }

    override fun onCreate(db: SQLiteDatabase) {
        // Création table Rapports
        val createRapportsTable = """
            CREATE TABLE $TABLE_RAPPORTS (
                $COL_ID INTEGER PRIMARY KEY AUTOINCREMENT,
                $COL_CODE TEXT UNIQUE NOT NULL,
                $COL_NOM TEXT NOT NULL,
                $COL_IMMAT TEXT,
                $COL_INVESTISSEMENT REAL DEFAULT 0,
                $COL_RECETTE REAL DEFAULT 0,
                $COL_DEPENSES REAL DEFAULT 0,
                $COL_SALAIRE REAL DEFAULT 0,
                $COL_COMMISSION REAL DEFAULT 0,
                $COL_RECETTE_NETTE REAL DEFAULT 0,
                $COL_DATE TEXT DEFAULT CURRENT_TIMESTAMP,
                $COL_SYNC INTEGER DEFAULT 0
            )
        """.trimIndent()
        
        // Création table Chauffeurs
        val createChauffeursTable = """
            CREATE TABLE $TABLE_CHAUFFEURS (
                $COL_CHAUFFEUR_ID INTEGER PRIMARY KEY AUTOINCREMENT,
                $COL_NOM_CHAUFFEUR TEXT NOT NULL,
                $COL_TELEPHONE TEXT,
                $COL_PERMIS TEXT UNIQUE NOT NULL,
                $COL_PHOTO_PERMIS TEXT,
                $COL_VEHICULE TEXT,
                $COL_DATE_EMBAUCHE TEXT,
                $COL_SALAIRE_BASE REAL DEFAULT 0,
                $COL_PRET_EN_COURS REAL DEFAULT 0,
                $COL_ACTIF INTEGER DEFAULT 1
            )
        """.trimIndent()
        
        db.execSQL(createRapportsTable)
        db.execSQL(createChauffeursTable)
    }

    override fun onUpgrade(db: SQLiteDatabase, oldVersion: Int, newVersion: Int) {
        db.execSQL("DROP TABLE IF EXISTS $TABLE_RAPPORTS")
        db.execSQL("DROP TABLE IF EXISTS $TABLE_CHAUFFEURS")
        onCreate(db)
    }

    // === OPERATIONS RAPPORTS ===
    
    fun ajouterRapport(rapport: Rapport): Long {
        val db = this.writableDatabase
        val values = ContentValues().apply {
            put(COL_CODE, rapport.codeProprietaire)
            put(COL_NOM, rapport.nomProprietaire)
            put(COL_IMMAT, rapport.immatriculation)
            put(COL_INVESTISSEMENT, rapport.investissementTotal)
            put(COL_RECETTE, rapport.recetteMensuelle)
            put(COL_DEPENSES, rapport.depensesPannes)
            put(COL_SALAIRE, rapport.salaireChauffeur)
            put(COL_COMMISSION, rapport.commissionGS)
            put(COL_RECETTE_NETTE, rapport.recetteNette)
            put(COL_SYNC, if (rapport.estSynchronise) 1 else 0)
        }
        val id = db.insert(TABLE_RAPPORTS, null, values)
        db.close()
        return id
    }

    fun getRapportsNonSynchronises(): List<Rapport> {
        val rapports = mutableListOf<Rapport>()
        val db = this.readableDatabase
        val cursor = db.rawQuery(
            "SELECT * FROM $TABLE_RAPPORTS WHERE $COL_SYNC = 0 ORDER BY $COL_DATE DESC", 
            null
        )
        
        if (cursor.moveToFirst()) {
            do {
                val rapport = Rapport(
                    id = cursor.getInt(cursor.getColumnIndexOrThrow(COL_ID)),
                    codeProprietaire = cursor.getString(cursor.getColumnIndexOrThrow(COL_CODE)),
                    nomProprietaire = cursor.getString(cursor.getColumnIndexOrThrow(COL_NOM)),
                    immatriculation = cursor.getString(cursor.getColumnIndexOrThrow(COL_IMMAT)),
                    investissementTotal = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_INVESTISSEMENT)),
                    recetteMensuelle = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_RECETTE)),
                    depensesPannes = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_DEPENSES)),
                    salaireChauffeur = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_SALAIRE)),
                    commissionGS = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_COMMISSION)),
                    recetteNette = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_RECETTE_NETTE)),
                    dateRapport = cursor.getString(cursor.getColumnIndexOrThrow(COL_DATE)),
                    estSynchronise = cursor.getInt(cursor.getColumnIndexOrThrow(COL_SYNC)) == 1
                )
                rapports.add(rapport)
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return rapports
    }

    fun marquerCommeSynchronise(id: Int) {
        val db = this.writableDatabase
        val values = ContentValues().apply {
            put(COL_SYNC, 1)
        }
        db.update(TABLE_RAPPORTS, values, "$COL_ID = ?", arrayOf(id.toString()))
        db.close()
    }

    // === OPERATIONS CHAUFFEURS ===
    
    fun ajouterChauffeur(chauffeur: Chauffeur): Long {
        val db = this.writableDatabase
        val values = ContentValues().apply {
            put(COL_NOM_CHAUFFEUR, chauffeur.nom)
            put(COL_TELEPHONE, chauffeur.telephone)
            put(COL_PERMIS, chauffeur.numeroPermis)
            put(COL_PHOTO_PERMIS, chauffeur.photoPermis)
            put(COL_VEHICULE, chauffeur.vehiculeId)
            put(COL_DATE_EMBAUCHE, chauffeur.dateEmbauche)
            put(COL_SALAIRE_BASE, chauffeur.salaireBase)
            put(COL_PRET_EN_COURS, chauffeur.pretEnCours)
            put(COL_ACTIF, if (chauffeur.estActif) 1 else 0)
        }
        val id = db.insert(TABLE_CHAUFFEURS, null, values)
        db.close()
        return id
    }

    fun getChauffeurs(): List<Chauffeur> {
        val chauffeurs = mutableListOf<Chauffeur>()
        val db = this.readableDatabase
        val cursor = db.rawQuery(
            "SELECT * FROM $TABLE_CHAUFFEURS ORDER BY $COL_NOM_CHAUFFEUR ASC", 
            null
        )
        
        if (cursor.moveToFirst()) {
            do {
                val chauffeur = Chauffeur(
                    id = cursor.getInt(cursor.getColumnIndexOrThrow(COL_CHAUFFEUR_ID)),
                    nom = cursor.getString(cursor.getColumnIndexOrThrow(COL_NOM_CHAUFFEUR)),
                    telephone = cursor.getString(cursor.getColumnIndexOrThrow(COL_TELEPHONE)),
                    numeroPermis = cursor.getString(cursor.getColumnIndexOrThrow(COL_PERMIS)),
                    photoPermis = cursor.getString(cursor.getColumnIndexOrThrow(COL_PHOTO_PERMIS)),
                    vehiculeId = cursor.getString(cursor.getColumnIndexOrThrow(COL_VEHICULE)),
                    dateEmbauche = cursor.getString(cursor.getColumnIndexOrThrow(COL_DATE_EMBAUCHE)),
                    salaireBase = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_SALAIRE_BASE)),
                    pretEnCours = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_PRET_EN_COURS)),
                    estActif = cursor.getInt(cursor.getColumnIndexOrThrow(COL_ACTIF)) == 1
                )
                chauffeurs.add(chauffeur)
            } while (cursor.moveToNext())
        }
        cursor.close()
        db.close()
        return chauffeurs
    }

    fun getChauffeurById(id: Int): Chauffeur? {
        val db = this.readableDatabase
        val cursor = db.rawQuery(
            "SELECT * FROM $TABLE_CHAUFFEURS WHERE $COL_CHAUFFEUR_ID = ?", 
            arrayOf(id.toString())
        )
        
        return if (cursor.moveToFirst()) {
            Chauffeur(
                id = cursor.getInt(cursor.getColumnIndexOrThrow(COL_CHAUFFEUR_ID)),
                nom = cursor.getString(cursor.getColumnIndexOrThrow(COL_NOM_CHAUFFEUR)),
                telephone = cursor.getString(cursor.getColumnIndexOrThrow(COL_TELEPHONE)),
                numeroPermis = cursor.getString(cursor.getColumnIndexOrThrow(COL_PERMIS)),
                photoPermis = cursor.getString(cursor.getColumnIndexOrThrow(COL_PHOTO_PERMIS)),
                vehiculeId = cursor.getString(cursor.getColumnIndexOrThrow(COL_VEHICULE)),
                dateEmbauche = cursor.getString(cursor.getColumnIndexOrThrow(COL_DATE_EMBAUCHE)),
                salaireBase = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_SALAIRE_BASE)),
                pretEnCours = cursor.getDouble(cursor.getColumnIndexOrThrow(COL_PRET_EN_COURS)),
                estActif = cursor.getInt(cursor.getColumnIndexOrThrow(COL_ACTIF)) == 1
            )
        } else {
            null
        }.also {
            cursor.close()
            db.close()
        }
    }

    fun supprimerChauffeur(id: Int): Int {
        val db = this.writableDatabase
        val result = db.delete(TABLE_CHAUFFEURS, "$COL_CHAUFFEUR_ID = ?", arrayOf(id.toString()))
        db.close()
        return result
    }
}

package com.digitalyourhope.gsservices.model

import java.util.Date

data class Rapport(
    val id: Int = 0,
    val codeProprietaire: String,
    val nomProprietaire: String,
    val immatriculation: String = "",
    val investissementTotal: Double = 0.0,
    val recetteMensuelle: Double = 0.0,
    val depensesPannes: Double = 0.0,
    val salaireChauffeur: Double = 0.0,
    val commissionGS: Double = 0.0,
    val recetteNette: Double = 0.0,
    val dateRapport: String = "",
    val estSynchronise: Boolean = false
) {
    // Calcul automatique du salaire chauffeur (25%)
    fun calculerSalaireChauffeur(): Double {
        return recetteMensuelle * 0.25
    }
    
    // Calcul automatique de la recette nette
    fun calculerRecetteNette(): Double {
        val salaire = if (salaireChauffeur == 0.0) calculerSalaireChauffeur() else salaireChauffeur
        return recetteMensuelle - depensesPannes - salaire - commissionGS
    }
    
    // Formater pour l'affichage
    fun getFormattedRecette(): String {
        return String.format("%,.0f FCFA", recetteMensuelle)
    }
}

package com.digitalyourhope.gsservices.model

data class Chauffeur(
    val id: Int = 0,
    val nom: String,
    val telephone: String = "",
    val numeroPermis: String,
    val photoPermis: String = "",
    val vehiculeId: String = "",
    val dateEmbauche: String = "",
    val salaireBase: Double = 0.0,
    val pretEnCours: Double = 0.0,
    val estActif: Boolean = true
) {
    // Calcul du salaire mensuel (25% de la recette du véhicule)
    fun calculerSalaireMensuel(recetteVehicule: Double): Double {
        return (recetteVehicule * 0.25) - pretEnCours
    }
    
    // Formater pour l'affichage
    fun getNomComplet(): String {
        return "Chauffeur $nom"
    }
}

package com.digitalyourhope.gsservices

import android.os.Bundle
import android.widget.Button

import com.digitalyourhope.gsservices.utils.DatabaseHelper
import kotlinx.coroutines.CoroutineScope

import okhttp3.*

import org.json.JSONObject
import java.io.IOException

class SyncActivity : AppCompatActivity() {

    private lateinit var progressBar: ProgressBar
    private lateinit var txtStatus: TextView
    private lateinit var txtStats: TextView
    private lateinit var btnSync: Button
    private lateinit var dbHelper: DatabaseHelper

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_sync)

        progressBar = findViewById(R.id.progressBar)
        txtStatus = findViewById(R.id.txtStatus)
        txtStats = findViewById(R.id.txtStats)
        btnSync = findViewById(R.id.btnSync)
        dbHelper = DatabaseHelper(this)

        afficherStats()
        btnSync.setOnClickListener {
            synchroniserDonnees()
        }
    }

    private fun afficherStats() {
        val rapportsNonSync = dbHelper.getRapportsNonSynchronises()
        val stats = "📊 En attente: ${rapportsNonSync.size} rapport(s)"
        txtStats.text = stats
    }

    private fun synchroniserDonnees() {
        val rapports = dbHelper.getRapportsNonSynchronises()
        
        if (rapports.isEmpty()) {
            txtStatus.text = "✅ Aucune donnée à synchroniser"
            txtStatus.setTextColor(resources.getColor(android.R.color.holo_green_dark))
            return
        }

        CoroutineScope(Dispatchers.IO).launch {
            synchroniserRapports(rapports)
        }
    }

    private suspend fun synchroniserRapports(rapports: List<com.digitalyourhope.gsservices.model.Rapport>) {
        withContext(Dispatchers.Main) {
            progressBar.visibility = android.view.View.VISIBLE
            progressBar.max = rapports.size
            txtStatus.text = "Synchronisation en cours..."
        }

        val client = OkHttpClient.Builder()
            .connectTimeout(30, java.util.concurrent.TimeUnit.SECONDS)
            .readTimeout(30, java.util.concurrent.TimeUnit.SECONDS)
            .build()

        var succes = 0
        var echecs = 0

        for ((index, rapport) in rapports.withIndex()) {
            val jsonData = JSONObject().apply {
                put("code_proprietaire", rapport.codeProprietaire)
                put("nom_proprietaire", rapport.nomProprietaire)
                put("immatriculation", rapport.immatriculation)
                put("investissement_total", rapport.investissementTotal)
                put("recette_mensuelle", rapport.recetteMensuelle)
                put("depenses_pannes", rapport.depensesPannes)
                put("salaire_chauffeur", rapport.calculerSalaireChauffeur())
                put("commission_gs", rapport.commissionGS)
                put("recette_nette", rapport.calculerRecetteNette())
            }

            val mediaType = "application/json; charset=utf-8".toMediaType()
            val body = jsonData.toString().toRequestBody(mediaType)

            val request = Request.Builder()
                .url("https://gsservices.page.gd/sync.php") // ← Ton URL réelle ici
                .post(body)
                .addHeader("User-Agent", "GS Services App")
                .build()

            try {
                val response = client.newCall(request).execute()
                val responseBody = response.body?.string()

                withContext(Dispatchers.Main) {
                    progressBar.progress = index + 1
                }

                if (response.isSuccessful && responseBody != null) {
                    val jsonResponse = JSONObject(responseBody)
                    if (jsonResponse.getString("status") == "success") {
                        dbHelper.marquerCommeSynchronise(rapport.id)
                        succes++
                    } else {
                        echecs++
                    }
                } else {
                    echecs++
                }
            } catch (e: IOException) {
                echecs++
            }
        }

        withContext(Dispatchers.Main) {
            progressBar.visibility = android.view.View.GONE
            
            val message = when {
                succes > 0 && echecs == 0 -> "✅ Synchronisation complète: $succès rapport(s)"
                succes > 0 && echecs > 0 -> "⚠ Synchronisation partielle: $succès OK, $echecs échec(s)"
                else -> "❌ Échec de la synchronisation"
            }
            
            txtStatus.text = message
            txtStatus.setTextColor(when {
                echecs == 0 -> resources.getColor(android.R.color.holo_green_dark)
                succes > 0 -> resources.getColor(android.R.color.holo_orange_dark)
                else -> resources.getColor(android.R.color.holo_red_dark)
            })
            
            afficherStats()
        }
    }
}

class ChauffeurAdapter(
    private var chauffeurs: List<Chauffeur>,
    private val onItemClick: (Chauffeur) -> Unit
) : RecyclerView.Adapter<ChauffeurAdapter.ViewHolder>() {
    
    class ViewHolder(view: android.view.View) : RecyclerView.ViewHolder(view) {
        val txtNom: TextView = view.findViewById(R.id.txtNomChauffeur)
        val txtPermis: TextView = view.findViewById(R.id.txtPermis)
        val txtVehicule: TextView = view.findViewById(R.id.txtVehicule)
        val txtStatut: TextView = view.findViewById(R.id.txtStatut)
    }
    
    override fun onCreateViewHolder(parent: android.view.viewGroup, viewType: Int): ViewHolder {
        val view = android.view.LayoutInflater.from(parent.context)
            .inflate(R.layout.item_chauffeur, parent, false)
        return ViewHolder(view)
    }
    
    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        val chauffeur = chauffeurs[position]
        
        holder.txtNom.text = chauffeur.nom
        holder.txtPermis.text = "Permis: ${chauffeur.numeroPermis}"
        holder.txtVehicule.text = "Véhicule: ${chauffeur.vehiculeId}"
        holder.txtStatut.text = if (chauffeur.estActif) "ACTIF" else "INACTIF"
        holder.txtStatut.setTextColor(android.graphics.Color.parseColor(
            if (chauffeur.estActif) "#4CAF50" else "#F44336"
        ))
        
        holder.itemView.setOnClickListener {
            onItemClick(chauffeur)
        }
    }
    
    override fun getItemCount() = chauffeurs.size
    
    fun updateData(newChauffeurs: List<Chauffeur>) {
        chauffeurs = newChauffeurs
        notifyDataSetChanged()
    }

import androidx.recyclerview.widget.LinearLayoutManager
import androidx.recyclerview.widget.RecyclerView
import com.digitalyourhope.gsservices.model.Chauffeur
import com.digitalyourhope.gsservices.utils.DatabaseHelper
import java.text.SimpleDateFormat
import java.util.*

class ChauffeursActivity : AppCompatActivity() {
    
    private lateinit var dbHelper: DatabaseHelper
    private lateinit var recyclerView: RecyclerView
    private lateinit var adapter: ChauffeurAdapter
    private lateinit var btnAjouter: Button
    private val REQUEST_IMAGE_CAPTURE = 1
    private var photoPermisPath: String = ""

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_chauffeurs)
        
        dbHelper = DatabaseHelper(this)
        recyclerView = findViewById(R.id.recyclerViewChauffeurs)
        btnAjouter = findViewById(R.id.btnAjouterChauffeur)
        
        // Configuration du RecyclerView
        recyclerView.layoutManager = LinearLayoutManager(this)
        adapter = ChauffeurAdapter(emptyList(), this::onChauffeurClick)
        recyclerView.adapter = adapter
        
        // Charger les chauffeurs
        chargerChauffeurs()
        
        btnAjouter.setOnClickListener {
            showAjouterChauffeurDialog()
        }
        
        // Bouton retour
        findViewById<Button>(R.id.btnRetour).setOnClickListener {
            finish()
        }
    }
    
    private fun chargerChauffeurs() {
        val chauffeurs = dbHelper.getChauffeurs()
        adapter.updateData(chauffeurs)
        
        val txtTotal = findViewById<TextView>(R.id.txtTotalChauffeurs)
        txtTotal.text = "Total: ${chauffeurs.size} chauffeur(s)"
    }
    
    private fun onChauffeurClick(chauffeur: Chauffeur) {
        showDetailsDialog(chauffeur)
    }
    
    private fun showAjouterChauffeurDialog() {
        val dialogView = layoutInflater.inflate(R.layout.dialog_ajouter_chauffeur, null)
        
        val editNom = dialogView.findViewById<EditText>(R.id.editNomChauffeur)
        val editTelephone = dialogView.findViewById<EditText>(R.id.editTelephone)
        val editPermis = dialogView.findViewById<EditText>(R.id.editPermis)
        val editVehicule = dialogView.findViewById<EditText>(R.id.editVehicule)
        val btnPhoto = dialogView.findViewById<Button>(R.id.btnPrendrePhoto)
        
        photoPermisPath = ""
        
        btnPhoto.setOnClickListener {
            prendrePhotoPermis()
        }
        
        AlertDialog.Builder(this)
            .setTitle("➕ NOUVEAU CHAUFFEUR")
            .setView(dialogView)
            .setPositiveButton("ENREGISTRER") { dialog, _ ->
                val nom = editNom.text.toString().trim()
                val permis = editPermis.text.toString().trim()
                
                if (nom.isEmpty() || permis.isEmpty()) {
                    Toast.makeText(this, "Nom et permis sont obligatoires", Toast.LENGTH_SHORT).show()
                } else {
                    val chauffeur = Chauffeur(
                        nom = nom,
                        telephone = editTelephone.text.toString().trim(),
                        numeroPermis = permis,
                        photoPermis = photoPermisPath,
                        vehiculeId = editVehicule.text.toString().trim(),
                        dateEmbauche = SimpleDateFormat("dd/MM/yyyy", Locale.FRANCE).format(Date())
                    )
                    
                    val id = dbHelper.ajouterChauffeur(chauffeur)
                    if (id > 0) {
                        Toast.makeText(this, "Chauffeur ajouté avec succès", Toast.LENGTH_SHORT).show()
                        chargerChauffeurs()
                    }
                }
                dialog.dismiss()
            }
            .setNegativeButton("ANNULER", null)
            .show()
    }
    
    private fun prendrePhotoPermis() {
        val intent = Intent(MediaStore.ACTION_IMAGE_CAPTURE)
        if (intent.resolveActivity(packageManager) != null) {
            startActivityForResult(intent, REQUEST_IMAGE_CAPTURE)
        } else {
            Toast.makeText(this, "Aucune app camera disponible", Toast.LENGTH_SHORT).show()
        }
    }
    
    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        super.onActivityResult(requestCode, resultCode, data)
        
        if (requestCode == REQUEST_IMAGE_CAPTURE && resultCode == Activity.RESULT_OK) {
            Toast.makeText(this, "Photo du permis prise", Toast.LENGTH_SHORT).show()
        }
    }
    
    private fun showDetailsDialog(chauffeur: Chauffeur) {
        AlertDialog.Builder(this)
            .setTitle("👨‍✈️ ${chauffeur.nom}")
            .setMessage(
                """
                Permis: ${chauffeur.numeroPermis}
                Téléphone: ${chauffeur.telephone}
                Véhicule: ${chauffeur.vehiculeId}
                Embauche: ${chauffeur.dateEmbauche}
                Prêt en cours: ${String.format("%,.0f", chauffeur.pretEnCours)} FCFA
                Statut: ${if (chauffeur.estActif) "ACTIF" else "INACTIF"}
                """.trimIndent()
            )
            .setPositiveButton("MODIFIER") { dialog, _ ->
                Toast.makeText(this, "Modification à venir", Toast.LENGTH_SHORT).show()
                dialog.dismiss()
            }
            .setNegativeButton("SUPPRIMER") { dialog, _ ->
                showConfirmSuppression(chauffeur.id)
                dialog.dismiss()
            }
            .setNeutralButton("PRÊT") { dialog, _ ->
                gererPret(chauffeur)
                dialog.dismiss()
            }
            .show()
    }
    
    private fun showConfirmSuppression(chauffeurId: Int) {
        AlertDialog.Builder(this)
            .setTitle("Confirmation")
            .setMessage("Voulez-vous vraiment supprimer ce chauffeur ?")
            .setPositiveButton("OUI") { dialog, _ ->
                val result = dbHelper.supprimerChauffeur(chauffeurId)
                if (result > 0) {
                    Toast.makeText(this, "Chauffeur supprimé", Toast.LENGTH_SHORT).show()
                    chargerChauffeurs()
                }
                dialog.dismiss()
            }
            .setNegativeButton("NON", null)
            .show()
    }
    
    private fun gererPret(chauffeur: Chauffeur) {
        Toast.makeText(this, "Gestion de prêt à implémenter", Toast.LENGTH_SHORT).show()
    }
}

// Adapter pour RecyclerView
class ChauffeurAdapter(
    private var chauffeurs: List<Chauffeur>,
    private val onItemClick: (Chauffeur) -> Unit
) : RecyclerView.Adapter<ChauffeurAdapter.ViewHolder>() {
    
    class ViewHolder(view: android.view.View) : RecyclerView.ViewHolder(view) {
        val txtNom: TextView = view.findViewById(R.id.txtNomChauffeur)
        val txtPermis: TextView = view.findViewById(R.id.txtPermis)
        val txtVehicule: TextView = view.findViewById(R.id.txtVehicule)
        val txtStatut: TextView = view.findViewById(R.id.txtStatut)
    }
    
    override fun onCreateViewHolder(parent: android.view.ViewGroup, viewType: Int): ViewHolder {
        val view = android.view.LayoutInflater.from(parent.context)
            .inflate(R.layout.item_chauffeur, parent, false)
        return ViewHolder(view)
    }
    
    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        val chauffeur = chauffeurs[position]
        
        holder.txtNom.text = chauffeur.nom
        holder.txtPermis.text = "Permis: ${chauffeur.numeroPermis}"
        holder.txtVehicule.text = "Véhicule: ${chauffeur.vehiculeId}"
        holder.txtStatut.text = if (chauffeur.estActif) "ACTIF" else "INACTIF"
        holder.txtStatut.setTextColor(android.graphics.Color.parseColor(
            if (chauffeur.estActif) "#4CAF50" else "#F44336"
        ))
        
        holder.itemView.setOnClickListener {
            onItemClick(chauffeur)
        }
    }
    
    override fun getItemCount() = chauffeurs.size
    
    fun updateData(newChauffeurs: List<Chauffeur>) {
        chauffeurs = newChauffeurs
        notifyDataSetChanged()
    }
}

package com.digitalyourhope.gsservices

import android.os.Bundle
import android.widget.Button
import android.widget.ProgressBar
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import com.digitalyourhope.gsservices.utils.DatabaseHelper

class SyncActivity : AppCompatActivity() {

    private lateinit var progressBar: ProgressBar
    private lateinit var txtStatus: TextView
    private lateinit var txtStats: TextView
    private lateinit var btnSync: Button
    private lateinit var dbHelper: DatabaseHelper

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_sync)

        progressBar = findViewById(R.id.progressBar)
        txtStatus = findViewById(R.id.txtStatus)
        txtStats = findViewById(R.id.txtStats)
        btnSync = findViewById(R.id.btnSync)
        dbHelper = DatabaseHelper(this)

        afficherStats()
        btnSync.setOnClickListener {
            simulerSynchronisation()
        }
    }

    private fun afficherStats() {
        val rapportsNonSync = dbHelper.getRapportsNonSynchronises()
        val stats = "📊 En attente: ${rapportsNonSync.size} rapport(s)"
        txtStats.text = stats
    }

    private fun simulerSynchronisation() {
        val rapports = dbHelper.getRapportsNonSynchronises()
        
        if (rapports.isEmpty()) {
            txtStatus.text = "✅ Aucune donnée à synchroniser"
            txtStatus.setTextColor(resources.getColor(android.R.color.holo_green_dark))
            return
        }

        progressBar.visibility = android.view.View.VISIBLE
        progressBar.max = rapports.size

                override fun run() {
                    progress++
                    progressBar.progress = progress
                    
                    if (progress <= rapports.size) {
                        handler.postDelayed(this, 500)
                    } else {
                        progressBar.visibility = android.view.View.GONE
                        
                        // Marquer comme synchronisé
                        rapports.forEach { rapport ->
                            dbHelper.marquerCommeSynchronise(rapport.id)
                        }

kotlin
package com.digitalyourhope.gsservices

import android.content.Intent
import android.os.Bundle
import androidx.appcompat.app.AppCompatActivity

class LaunchActivity : AppCompatActivity() {
    
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        // Ici tu peux décider quelle activité lancer en premier
        // Option A: Lancer TestActivity pour les tests
        // Option B: Lancer MainActivity directement
        
        val intent: Intent
        
        // MODIFIER CETTE LIGNE POUR CHOISIR:
        // true = démarrer sur TestActivity (pour développement)
        // false = démarrer sur MainActivity (pour production)
        val demarrerAvecTests = true
        
        if (demarrerAvecTests) {
            intent = Intent(this, TestActivity::class.java)
        } else {
            intent = Intent(this, MainActivity::class.java)
        }
        
        // Lancer l'activité choisie
        startActivity(intent)
        finish() // Fermer LaunchActivity
    }
}

// Réponse par défaut
$response = [

    'timestamp' => date('Y-m-d H:i:s')

// Vérifier la méthode HTTP
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    // Récupérer les données JSON
    $json = file_get_contents('php://input');
    $data = json_decode($json, true);
    
    if ($data) {
        try {
            // Connexion à la base de données
            $conn = new mysqli(DB_HOST, DB_USER, DB_PASS, DB_NAME);
            
            if ($conn->connect_error) {
                throw new Exception("Erreur connexion DB: " . $conn->connect_error);
            }
            
            // Préparer les données
            $code = $conn->real_escape_string($data['code_proprietaire'] ?? '');
            $nom = $conn->real_escape_string($data['nom_proprietaire'] ?? '');
            $immat = $conn->real_escape_string($data['immatriculation'] ?? '');
            $recette = floatval($data['recette_mensuelle'] ?? 0);
            $depenses = floatval($data['depenses_pannes'] ?? 0);
            $commission = floatval($data['commission_gs'] ?? ($recette * 0.10));
            $salaire_chauffeur = $recette * 0.25; // 25% automatique
            $recette_nette = $recette - $depenses - $salaire_chauffeur - $commission;
            
            // Vérifier si le rapport existe déjà ce mois-ci
            $sql_check = "SELECT id FROM rapports 
                         WHERE code_proprietaire = '$code' 
                         AND MONTH(date_rapport) = MONTH(CURRENT_DATE())
                         AND YEAR(date_rapport) = YEAR(CURRENT_DATE())";
            
            $result = $conn->query($sql_check);
            
            if ($result->num_rows > 0) {
                // Mettre à jour le rapport existant
                $sql = "UPDATE rapports SET 
                        recette_mensuelle = '$recette',
                        depenses_pannes = '$depenses',
                        salaire_chauffeur = '$salaire_chauffeur',
                        commission_gs = '$commission',
                        recette_nette = '$recette_nette'
                        WHERE code_proprietaire = '$code' 
                        AND MONTH(date_rapport) = MONTH(CURRENT_DATE())";
            } else {
                // Insérer un nouveau rapport
                $sql = "INSERT INTO rapports 
                        (code_proprietaire, nom_proprietaire, immatriculation, 
                         recette_mensuelle, depenses_pannes, salaire_chauffeur, 
                         commission_gs, recette_nette) 
                        VALUES 
                        ('$code', '$nom', '$immat', 
                         '$recette', '$depenses', '$salaire_chauffeur',
                         '$commission', '$recette_nette')";
            }
            
            // Exécuter la requête
            if ($conn->query($sql) === TRUE) {
                $response['status'] = 'success';
                $response['message'] = 'Données synchronisées avec succès';
                $response['id'] = $conn->insert_id;
                $response['calculs'] = [
                    'salaire_chauffeur' => $salaire_chauffeur,
                    'recette_nette' => $recette_nette
                ];
            } else {
                throw new Exception("Erreur SQL: " . $conn->error);
            }
            
            $conn->close();
            
        } catch (Exception $e) {
            $response['message'] = $e->getMessage();
            if (DEBUG_MODE) {
                $response['debug'] = [
                    'data_received' => $data,
                    'sql' => $sql ?? 'N/A'
                ];
            }
        }
    }
}

// Retourner la réponse en JSON
echo json_encode($response, JSON_PRETTY_PRINT);

package com.digitalyourhope.gsservices

import android.os.Bundle
import android.widget.Button
import android.widget.ProgressBar
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import com.digitalyourhope.gsservices.utils.DatabaseHelper
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONObject
import java.io.IOException

class SyncActivity : AppCompatActivity() {

    private lateinit var progressBar: ProgressBar
    private lateinit var txtStatus: TextView
    private lateinit var txtStats: TextView
    private lateinit var btnSync: Button
    private lateinit var dbHelper: DatabaseHelper

    // ⚠️ REMPLACE PAR TON URL INFINITYFREE
    private val SERVER_URL = "https://gsservices.page.gd/sync.php"

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_sync)

        progressBar = findViewById(R.id.progressBar)
        txtStatus = findViewById(R.id.txtStatus)
        txtStats = findViewById(R.id.txtStats)
        btnSync = findViewById(R.id.btnSync)
        dbHelper = DatabaseHelper(this)

        afficherStats()
        btnSync.setOnClickListener {
            synchroniserAvecServeur()
        }
    }

    private fun afficherStats() {
        val rapportsNonSync = dbHelper.getRapportsNonSynchronises()
        val stats = "📊 En attente: ${rapportsNonSync.size} rapport(s)"
        txtStats.text = stats
    }

    private fun synchroniserAvecServeur() {
        val rapports = dbHelper.getRapportsNonSynchronises()
        
        if (rapports.isEmpty()) {
            txtStatus.text = "✅ Aucune donnée à synchroniser"
            txtStatus.setTextColor(resources.getColor(android.R.color.holo_green_dark))
            return
        }

        CoroutineScope(Dispatchers.IO).launch {
            synchroniserRapportsReels(rapports)
        }
    }

    private suspend fun synchroniserRapportsReels(rapports: List<com.digitalyourhope.gsservices.model.Rapport>) {
        withContext(Dispatchers.Main) {
            progressBar.visibility = android.view.View.VISIBLE
            progressBar.max = rapports.size
            progressBar.progress = 0
            txtStatus.text = "Connexion au serveur..."
        }

        val client = OkHttpClient.Builder()
            .connectTimeout(30, java.util.concurrent.TimeUnit.SECONDS)
            .readTimeout(30, java.util.concurrent.TimeUnit.SECONDS)
            .build()

        var succes = 0
        var echecs = 0

        for ((index, rapport) in rapports.withIndex()) {
            // Préparer les données JSON
            val jsonData = JSONObject().apply {
                put("code_proprietaire", rapport.codeProprietaire)
                put("nom_proprietaire", rapport.nomProprietaire)
                put("immatriculation", rapport.immatriculation)
                put("recette_mensuelle", rapport.recetteMensuelle)
                put("depenses_pannes", rapport.depensesPannes)
                put("commission_gs", rapport.commissionGS)
            }

            val mediaType = "application/json; charset=utf-8".toMediaType()
            val body = jsonData.toString().toRequestBody(mediaType)

            val request = Request.Builder()
                .url(SERVER_URL)
                .post(body)
                .addHeader("User-Agent", "GS Services App")
                .addHeader("Content-Type", "application/json")
                .build()

            try {
                val response = client.newCall(request).execute()
                val responseBody = response.body?.string()

                withContext(Dispatchers.Main) {
                    progressBar.progress = index + 1
                    txtStatus.text = "Synchronisation ${index + 1}/${rapports.size}"
                }

                if (response.isSuccessful && responseBody != null) {
                    val jsonResponse = JSONObject(responseBody)
                    if (jsonResponse.getString("status") == "success") {
                        dbHelper.marquerCommeSynchronise(rapport.id)
                        succes++
                    } else {
                        echecs++
                    }
                } else {
                    echecs++
                }
            } catch (e: IOException) {
                echecs++
                withContext(Dispatchers.Main) {
                    txtStatus.text = "❌ Erreur de connexion"
                }
            } catch (e: Exception) {
                echecs++
            }
        }

        withContext(Dispatchers.Main) {
            progressBar.visibility = android.view.View.GONE
            
            val message = when {
                succes > 0 && echecs == 0 -> "✅ Synchronisation réussie: $succès rapport(s)"
                succes > 0 && echecs > 0 -> "⚠ Synchronisation partielle: $succès OK, $echecs échec(s)"
                else -> "❌ Échec de la synchronisation"
            }
            
            txtStatus.text = message
            txtStatus.setTextColor(when {
                echecs == 0 -> resources.getColor(android.R.color.holo_green_dark)
                succes > 0 -> resources.getColor(android.R.color.holo_orange_dark)
                else -> resources.getColor(android.R.color.holo_red_dark)
            })
            
            afficherStats()
        }
    }
}

kotlin
package com.digitalyourhope.gsservices

import android.os.Bundle
import android.widget.Button
import android.widget.ProgressBar
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import com.digitalyourhope.gsservices.utils.DatabaseHelper
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONObject
import java.io.IOException

class SyncActivity : AppCompatActivity() {

    private lateinit var progressBar: ProgressBar
    private lateinit var txtStatus: TextView
    private lateinit var txtStats: TextView
    private lateinit var btnSync: Button
    private lateinit var dbHelper: DatabaseHelper

    // ⚠️ REMPLACE PAR TON URL INFINITYFREE
    private val SERVER_URL = "https://gsservices.page.gd/sync.php"

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_sync)

        progressBar = findViewById(R.id.progressBar)
        txtStatus = findViewById(R.id.txtStatus)
        txtStats = findViewById(R.id.txtStats)
        btnSync = findViewById(R.id.btnSync)
        dbHelper = DatabaseHelper(this)

        afficherStats()
        btnSync.setOnClickListener {
            synchroniserAvecServeur()
        }
    }

    private fun afficherStats() {
        val rapportsNonSync = dbHelper.getRapportsNonSynchronises()
        val stats = "📊 En attente: ${rapportsNonSync.size} rapport(s)"
        txtStats.text = stats
    }

    private fun synchroniserAvecServeur() {
        val rapports = dbHelper.getRapportsNonSynchronises()
        
        if (rapports.isEmpty()) {
            txtStatus.text = "✅ Aucune donnée à synchroniser"
            txtStatus.setTextColor(resources.getColor(android.R.color.holo_green_dark))
            return
        }

        CoroutineScope(Dispatchers.IO).launch {
            synchroniserRapportsReels(rapports)
        }
    }

    private suspend fun synchroniserRapportsReels(rapports: List<com.digitalyourhope.gsservices.model.Rapport>) {
        withContext(Dispatchers.Main) {
            progressBar.visibility = android.view.View.VISIBLE
            progressBar.max = rapports.size
            progressBar.progress = 0
            txtStatus.text = "Connexion au serveur..."
        }

        val client = OkHttpClient.Builder()
            .connectTimeout(30, java.util.concurrent.TimeUnit.SECONDS)
            .readTimeout(30, java.util.concurrent.TimeUnit.SECONDS)
            .build()

        var successCount = 0  // Changé de 'succes' à 'successCount'
        var echecs = 0

        for ((index, rapport) in rapports.withIndex()) {
            // Préparer les données JSON
            val jsonData = JSONObject().apply {
                put("code_proprietaire", rapport.codeProprietaire)
                put("nom_proprietaire", rapport.nomProprietaire)
                put("immatriculation", rapport.immatriculation)
                put("recette_mensuelle", rapport.recetteMensuelle)
                put("depenses_pannes", rapport.depensesPannes)
                put("commission_gs", rapport.commissionGS)
            }

            val mediaType = "application/json; charset=utf-8".toMediaType()
            val body = jsonData.toString().toRequestBody(mediaType)

            val request = Request.Builder()
                .url(SERVER_URL)
                .post(body)
                .addHeader("User-Agent", "GS Services App")
                .addHeader("Content-Type", "application/json")
                .build()

            try {
                val response = client.newCall(request).execute()
                val responseBody = response.body?.string()

                withContext(Dispatchers.Main) {
                    progressBar.progress = index + 1
                    txtStatus.text = "Synchronisation ${index + 1}/${rapports.size}"
                }

                if (response.isSuccessful && responseBody != null) {
                    val jsonResponse = JSONObject(responseBody)
                    if (jsonResponse.getString("status") == "success") {
                        dbHelper.marquerCommeSynchronise(rapport.id)
                        successCount++  // Utilisation de la variable corrigée
                    } else {
                        echecs++
                    }
                } else {
                    echecs++
                }
            } catch (e: IOException) {
                echecs++
                withContext(Dispatchers.Main) {
                    txtStatus.text = "❌ Erreur de connexion"
                }
            } catch (e: Exception) {
                echecs++
            }
        }

        withContext(Dispatchers.Main) {
            progressBar.visibility = android.view.View.GONE
            
            val message = when {
                successCount > 0 && echecs == 0 -> "✅ Synchronisation réussie: $successCount rapport(s)"
                successCount > 0 && echecs > 0 -> "⚠ Synchronisation partielle: $successCount OK, $echecs échec(s)"
                else -> "❌ Échec de la synchronisation"
            }
            
            txtStatus.text = message
            txtStatus.setTextColor(when {
                echecs == 0 -> resources.getColor(android.R.color.holo_green_dark)
                successCount > 0 -> resources.getColor(android.R.color.holo_orange_dark)
                else -> resources.getColor(android.R.color.holo_red_dark)
            })
            
            afficherStats()
        }
    }
}

class ChauffeurAdapter(

        val txtVehicule: TextView = view.findViewById(R.id.txtVehicule)

    }

        val view = android.view.LayoutInflater.from(parent.context)
            .inflate(R.layout.item_chauffeur, parent, false)
        return ViewHolder(view)
    }
    
    override fun onBindViewHolder(holder: ViewHolder, position: Int) {
        val chauffeur = chauffeurs[position]
        
        holder.txtNom.text = chauffeur.nom
        holder.txtPermis.text = "Permis: ${chauffeur.numeroPermis}"

FICHIER 1 : LaunchManager.kt (Nouveau fichier pour gérer le lancement)
kotlin
package com.digitalyourhope.gsservices.utils

import android.content.Context
import android.content.SharedPreferences

class LaunchManager(context: Context) {
    
    private val prefs: SharedPreferences = context.getSharedPreferences("gs_prefs", Context.MODE_PRIVATE)
    
    companion object {
        const val KEY_FIRST_LAUNCH = "first_launch"
        const val KEY_TESTS_COMPLETED = "tests_completed"
    }
    
    // Vérifier si c'est le premier lancement
    fun isFirstLaunch(): Boolean {
        return prefs.getBoolean(KEY_FIRST_LAUNCH, true)
    }
    
    // Marquer que ce n'est plus le premier lancement
    fun markFirstLaunchCompleted() {
        prefs.edit().putBoolean(KEY_FIRST_LAUNCH, false).apply()
    }
    
    // Vérifier si les tests sont terminés
    fun areTestsCompleted(): Boolean {
        return prefs.getBoolean(KEY_TESTS_COMPLETED, false)
    }
    
    // Marquer les tests comme terminés
    fun markTestsCompleted() {
        prefs.edit().putBoolean(KEY_TESTS_COMPLETED, true).apply()
    }
    
    // Réinitialiser pour les tests de développement
    fun resetForTesting() {
        prefs.edit().clear().apply()
    }
}
FICHIER 2 : MainActivity.kt (VERSION FINALE CORRIGÉE)
kotlin
package com.digitalyourhope.gsservices

import android.content.Intent
import android.os.Bundle
import android.widget.Button
import androidx.appcompat.app.AppCompatActivity
import com.digitalyourhope.gsservices.utils.LaunchManager

class MainActivity : AppCompatActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        // Initialiser le gestionnaire de lancement
        val launchManager = LaunchManager(this)
        
        // LOGIQUE DE LANCEMENT INTELLIGENTE
        if (launchManager.isFirstLaunch() && !launchManager.areTestsCompleted()) {
            // PREMIER LANCEMENT : Aller à TestActivity
            launchManager.markFirstLaunchCompleted()
            val intent = Intent(this, TestActivity::class.java)
            startActivity(intent)
            finish()
            return
        }
        
        // LANCEMENT NORMAL : Afficher l'interface principale
        setContentView(R.layout.activity_main)
        initialiserInterface()
    }
    
    private fun initialiserInterface() {
        // Afficher les boutons IMMÉDIATEMENT
        val buttonsLayout = findViewById<android.view.View>(R.id.buttonsLayout)
        buttonsLayout.visibility = android.view.View.VISIBLE
        buttonsLayout.alpha = 1f

        // Bouton Administrateur
        val btnAdmin = findViewById<Button>(R.id.btnAdmin)
        btnAdmin.setOnClickListener {
            val intent = Intent(this, AdminActivity::class.java)
            startActivity(intent)
        }

        // Bouton Client
        val btnClient = findViewById<Button>(R.id.btnClient)
        btnClient.setOnClickListener {
            // Ouvre le site web dans le navigateur
            val webIntent = Intent(Intent.ACTION_VIEW)
            webIntent.data = android.net.Uri.parse("https://gsservices.page.gd")
            startActivity(webIntent)
        }
    }
}

package com.digitalyourhope.gsservices

        btnTestRapport.setOnClickListener { testerRapport() }
        btnTestChauffeur.setOnClickListener { testerChauffeur() }
        btnLancerApp.setOnClickListener { 
            lancerApplication()
        }
        btnSkipTests.setOnClickListener {
            skipTestsAndLaunch()
        }
        
        // Message de bienvenue
        txtResultat.text = """
            🔍 DIGITAL YOURHOPE - TESTS GS SERVICES
            
            Cette écran apparaît uniquement au premier lancement.
            
            Tests recommandés:
            1. Base de données
            2. Rapports avec calculs automatiques
            3. Gestion des chauffeurs
            
            Après les tests, cliquez sur "LANCER L'APPLICATION".
            
            Pour passer directement à l'app, cliquez sur "PASSER LES TESTS".
        """.trimIndent()
    }
    
    private fun testerBaseDonnees() {
        try {
            // Test de création de la base
            val db = dbHelper.writableDatabase
            
            txtResultat.text = """
                ✅ BASE DE DONNÉES CRÉÉE
                
                Test réussi !
                Base: ${db.path}
                
                Prochaine étape: Testez les rapports et chauffeurs.
            """.trimIndent()

        }
    }

        try {
            // Créer un rapport test

                ✅ TEST RAPPORT RÉUSSI
                
                ID inséré: $id
                
                Calculs automatiques:
                • Recette: ${String.format("%,.0f", rapport.recetteMensuelle)} FCFA
                • Salaire chauffeur (25%): ${String.format("%,.0f", rapport.calculerSalaireChauffeur())} FCFA

                ✅ Calculs automatiques fonctionnent !
            """.trimIndent()
                
        } catch (e: Exception) {
            txtResultat.text = "❌ ERREUR RAPPORT: ${e.message}"
        }
    }
    
    private fun testerChauffeur() {
        try {
            // Créer un chauffeur test
            val chauffeur = Chauffeur(
                nom = "Jean Testeur",
                telephone = "06 123 45 67",

    private fun lancerApplication() {
        Toast.makeText(this, "Tests terminés. Lancement de l'application...", Toast.LENGTH_SHORT).show()
        txtResultat.text = "🔄 Tests terminés avec succès !\n\nLancement de GS Services..."
        
        // Marquer les tests comme terminés
        launchManager.markTestsCompleted()
        
        // Donner un peu de temps pour afficher le message
        android.os.Handler(android.os.Looper.getMainLooper()).postDelayed({
            try {
                // Fermer la base de données
                dbHelper.close()
                
                // Lancer MainActivity SANS créer de boucle
                val intent = Intent(this@TestActivity, MainActivity::class.java)
                
                // Flags pour éviter les problèmes
                intent.flags = Intent.FLAG_ACTIVITY_CLEAR_TASK or Intent.FLAG_ACTIVITY_NEW_TASK
                
                // Démarrer et fermer
                startActivity(intent)
                finish()
                
            } catch (e: Exception) {
                txtResultat.text = "❌ Erreur: ${e.message}"
                Toast.makeText(this, "Erreur: ${e.message}", Toast.LENGTH_LONG).show()
            }
        }, 1500)
    }
    
    private fun skipTestsAndLaunch() {
        Toast.makeText(this, "Passage direct à l'application...", Toast.LENGTH_SHORT).show()
        txtResultat.text = "⏩ Passage direct à l'application..."
        
        // Marquer les tests comme terminés
        launchManager.markTestsCompleted()
        
        android.os.Handler(android.os.Looper.getMainLooper()).postDelayed({
            try {
                // Lancer MainActivity directement
                val intent = Intent(this@TestActivity, MainActivity::class.java)
                intent.flags = Intent.FLAG_ACTIVITY_CLEAR_TASK or Intent.FLAG_ACTIVITY_NEW_TASK
                startActivity(intent)
                finish()
            } catch (e: Exception) {
                txtResultat.text = "❌ Erreur: ${e.message}"
            }
        }, 1000)
    }
    
    override fun onBackPressed() {
        // Empêcher de retourner en arrière vers MainActivity
        moveTaskToBack(true)
    }
}
